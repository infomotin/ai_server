Using Skills via API

Reference for integrating skills into your applications

# Python - using OpenAI SDK
import openai

openai.api_key = "sk-local-your-key"
openai.api_base = "http://203.55.176.101:8000/v1"

# Use a skill by ID in chat completions
response = openai.ChatCompletion.create(
    model="llama3.2:1b",
    messages=[{"role": "user", "content": "Review this code..."}],
    extra_body={"skill_id": "YOUR_SKILL_ID"}
)

# Or use the skill-enhanced endpoint directly
response = requests.post(
    "http://203.55.176.101:8000/v1/chat/completions/skill",
    json={
        "model": "llama3.2:1b",
        "messages": [{"role": "user", "content": "..."}],
        "skill_name": "Code Reviewer"
    },
    headers={"Authorization": "Bearer sk-local-..."}
)
